local EquipmentSet = require("Gameplay.Items.EquipmentSet")
local EquipmentEffect = require("Gameplay.Items.EquipmentEffect")

local EquipmentSets = {
    equipmentSets = {}
}

---注册套装
---@param id string
---@param name string
---@param twoPieceBonus EquipmentEffect
---@param fourPieceBonus EquipmentEffect
---@param sixPieceBonus EquipmentEffect
---@return EquipmentSet
function EquipmentSets:Register(id, name, twoPieceBonus, fourPieceBonus, sixPieceBonus)
    local set = EquipmentSet(id, name, twoPieceBonus, fourPieceBonus, sixPieceBonus)
    self.equipmentSets[id] = set
    return set
end

---
---@param id string
---@return EquipmentSet
function EquipmentSets:Get(id)
    return self.equipmentSets[id]
end
