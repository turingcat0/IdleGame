---@class MetaSave 元数据存档模块（总游戏时长、死亡次数、生涯战斗统计）
local MetaSave = {}

---收集元数据
---@param context table
---@return table
function MetaSave.Collect(context)
    local gameManager = context.gameManager
    local career = gameManager.combatStatsTracker:GetCareerStats()
    return {
        totalPlayTime = gameManager.totalPlayTime or 0,
        deathCount = gameManager.deathCount or 0,
        combat = {
            effectsAdded = career.effectsAdded,
            effectsRemoved = career.effectsRemoved,
            effectStackChanges = career.effectStackChanges,
            effectDamageDealt = career.effectDamageDealt,
            effectDamageAbsorbed = career.effectDamageAbsorbed,
        },
    }
end

---将存档数据应用到元数据系统
---@param context table
---@param data table
function MetaSave.Apply(context, data)
    local gameManager = context.gameManager
    gameManager.totalPlayTime = tonumber(data.totalPlayTime) or 0
    gameManager.deathCount = tonumber(data.deathCount) or 0
    if gameManager.combatStatsTracker and data.combat then
        gameManager.combatStatsTracker:LoadCareerStats(data.combat)
    end
end

---默认元数据
---@return table
function MetaSave.Default()
    return {
        totalPlayTime = 0,
        deathCount = 0,
        combat = {
            effectsAdded = {},
            effectsRemoved = {},
            effectStackChanges = 0,
            effectDamageDealt = 0,
            effectDamageAbsorbed = 0,
        },
    }
end

return MetaSave
