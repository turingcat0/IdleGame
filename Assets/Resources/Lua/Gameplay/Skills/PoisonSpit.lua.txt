local Skill = require("Gameplay.Skills.Skill")
local AttributeType = require("Gameplay.Stats.AttributeType")
local EffectTypes = require("Config.Generated.EffectTypes")
local EffectFactory = require("Gameplay.Effects.EffectFactory")

---@class PoisonSpit : Skill
---毒液喷射：命中时给目标上毒（攻击力 15% DPS，4秒）
local PoisonSpit = Skill:extend()

function PoisonSpit:OnCast(source, target)
end

function PoisonSpit:OnHit(source, target, ctx)
    if not target then return end
    local attackAttr = source.attributes and source.attributes[AttributeType.Attack]
    local attack = attackAttr and attackAttr:GetFinalValue() or 10
    local dps = attack * 0.15
    local instance = EffectFactory.CreateInstance(EffectTypes.poison, target, source, 4, { dps = dps })
    target:AddEffect(instance)
end

return PoisonSpit
