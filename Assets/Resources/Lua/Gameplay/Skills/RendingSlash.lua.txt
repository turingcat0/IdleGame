local Skill = require("Gameplay.Skills.Skill")
local EffectTypes = require("Config.Generated.EffectTypes")
local EffectFactory = require("Gameplay.Effects.EffectFactory")

---@class RendingSlash : Skill
---裂斩：命中时降低目标防御（30%，3秒）
local RendingSlash = Skill:extend()

function RendingSlash:OnCast(source, target)
end

---@param source Entity
---@param target Entity
---@param ctx DamageContext
function RendingSlash:OnHit(source, target, ctx)
    if not target then return end
    local instance = EffectFactory.CreateInstance(EffectTypes.defense_down, target, source, 3, { value = -0.3 })
    target:AddEffect(instance)
end

return RendingSlash
