local Skill = require("Gameplay.Skills.Skill")
local EffectTypes = require("Config.Generated.EffectTypes")
local EffectFactory = require("Gameplay.Effects.EffectFactory")

---@class ElderSmash : Skill
---熔岩重击：命中时降低目标防御（30%，4秒）
local ElderSmash = Skill:extend()

function ElderSmash:OnCast(source, target)
end

function ElderSmash:OnHit(source, target, ctx)
    if not target then return end
    local instance = EffectFactory.CreateInstance(EffectTypes.defense_down, target, source, 4, { value = -0.3 })
    target:AddEffect(instance)
end

return ElderSmash
