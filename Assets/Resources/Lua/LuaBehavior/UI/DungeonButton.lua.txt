---@type CS.UnityEngine.MonoBehaviour
local self = self

---@type CS.UnityEngine.UI.Button
local button = nil

function Start()
    button = self:GetComponent(typeof(CS.UnityEngine.UI.Button))
    button.onClick:AddListener(OnButtonClicked)
end

function OnDestroy()
    button.onClick:RemoveListener(OnButtonClicked)
end

function OnButtonClicked()
    local windowGo = CS.UnityEngine.GameObject.Find("DungeonWindow")
    if not windowGo then
        print("[DungeonButton] 找不到 DungeonWindow")
        return
    end

    local lb = windowGo:GetComponent(typeof(CS.LuaBehaviour))
    if lb and lb.scriptEnv then
        lb.scriptEnv.Show()
    end
end
