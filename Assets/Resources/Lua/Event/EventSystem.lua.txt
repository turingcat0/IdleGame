local EventType = require("Event.EventType")
---@class EventSystem
---@field listeners table<EventType ,table<any, {object: any, listener: function}[]>>
local EventSystem = {
    listeners = {}
}


---通用的监听器注册
---@param event EventType
---@param target any? target为空时则所有同类事件都会调用这个触发器
---@param object any
---@param listener function
function EventSystem:RegisterListener(event, target, object, listener)
    local eventTable = self.listeners[event]
    if not eventTable then
        eventTable = {}
        self.listeners[event] = eventTable
    end
    if not target then
        target = self
    end
    local eventObjectTable = eventTable[target]
    if not eventObjectTable then
        eventObjectTable = {}
        eventTable[target] = eventObjectTable
    end

    table.insert(eventObjectTable, {object = object, listener = listener})
end

---通用的触发事件
---@param event EventType
---@param target any
---@param ... any
function EventSystem:FireEvent(event, target, ...)
    local eventTable = self.listeners[event]
    if eventTable then
        local eventObjectTable = eventTable[target]
        if eventObjectTable then
            for _, pair in ipairs(eventObjectTable) do
                if pair.object then
                    pair.listener(pair.object, ...)
                else
                    pair.listener(...)
                end
            end
        end
        eventObjectTable = eventTable[self]
        if eventObjectTable then
            for _, pair in ipairs(eventObjectTable) do
                if pair.object then
                    pair.listener(pair.object, ...)
                else
                    pair.listener(...)
                end
            end
        end
    end
end

---通用的移除监听器
---@param event EventType
---@param target any?
---@param object any
---@param listener function
function EventSystem:RemoveListener(event, target, object, listener)
    if not target then
        target = self
    end
    --既然都是移除了，没加过的listener移除就让它爆掉好了
    for i, pair in ipairs(self.listeners[event][target]) do
        if pair.listener == listener and pair.object == object then
            table.remove(self.listeners[event][target], i)
            return
        end
    end
end





---注册属性变动监听器
---@param entity Entity
---@param attributeType AttributeType
---@param object any
---@param listener function
function EventSystem:RegisterAttributeChangedListener(entity, attributeType, object, listener)
    self:RegisterListener(EventType.AttributeChanged, entity.attributes[attributeType], object, listener)
end


---触发属性变动事件
---@param attribute Attribute
function EventSystem:FireAttributeChangedEvent(attribute)
    self:FireEvent(EventType.AttributeChanged, attribute)
end

---移除属性变动监听器

---@param object any
---@param listener function
function EventSystem:RemoveAttributeChangedListener(entity, attributeType, object, listener)

end
